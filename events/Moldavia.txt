	id = moldavia.1
	title = moldavia.1.t
	desc = moldavia.1.d

		tag = MOL

			OR = {
				is_puppet_of = ROM
				has_autonomy_state = autonomy_dominion
			}
		}
		NOT = { has_war_with = ROM }
	}
	mean_time_to_happen = {
		days = 1
	}
	fire_only_once = yes
	
	option = {
		name = moldavia.1.a
		ai_chance = {
			factor = 100
		}
		
		# FIX: не retiring лидера правящей партии
		if = {
			limit = { NOT = { has_government = democratic } }
			retire_ideology_leader = democratic
		}
		if = {
			limit = { NOT = { has_government = fascism } }
			retire_ideology_leader = fascism
		}
		if = {
			limit = { NOT = { has_government = neutrality } }
			retire_ideology_leader = neutrality
		}

		set_party_name = {
			ideology = democratic

			name = Free_MOL_democratic_party
